<script>
const ROOM_NUMBERS = [
  101,102,
  103,104,
  105,106,
  107,108
];

let rooms = [];

let score = 0;
let cleaned = 0;
let happy = 0;

/*
0 = максимально довольны
100 = проигрыш
*/
let dissatisfaction = 12;

let gameMinutes = 8 * 60;

let running = false;
let cleaning = false;

let mainTimer = null;
let eventTimer = null;

const clockEl = document.getElementById("clock");
const scoreEl = document.getElementById("score");
const cleanedEl = document.getElementById("cleaned");
const happyEl = document.getElementById("happy");

const pointer = document.getElementById("pointer");
const satisfactionBox =
  document.getElementById("satisfactionBox");

const jobMain = document.getElementById("jobMain");
const jobBar = document.getElementById("jobBar");
const message = document.getElementById("message");

const cart = document.getElementById("cart");

const leftRooms =
  document.getElementById("leftRooms");

const rightRooms =
  document.getElementById("rightRooms");

const startScreen =
  document.getElementById("startScreen");

const gameOverScreen =
  document.getElementById("gameOver");


function createRooms(){

  rooms = ROOM_NUMBERS.map((number,index)=>({
    number,
    index,
    status:"ready",
    waiting:0
  }));

  renderRooms();
}


function renderRooms(){

  leftRooms.innerHTML="";
  rightRooms.innerHTML="";

  rooms.forEach((room,index)=>{

    const el=document.createElement("div");

    el.className="room "+room.status;

    let icon="🟢";
    let text="ГОТОВ";

    if(room.status==="clean"){
      icon="🧹";
      text="ТРЕБУЕТ УБОРКИ";
    }

    if(room.status==="urgent"){
      icon="⚠️";
      text="СРОЧНО";
    }

    if(room.status==="dnd"){
      icon="⛔";
      text="НЕ БЕСПОКОИТЬ";
    }

    el.innerHTML=`
      <div class="number">${room.number}</div>
      <div class="roomIcon">${icon}</div>
      <div class="roomText">${text}</div>

      ${
        room.status==="clean" ||
        room.status==="urgent"
        ?
        `<div class="roomProgress">
           <div style="width:${Math.min(room.waiting,100)}%"></div>
         </div>`
        :""
      }
    `;

    el.addEventListener("click",()=>{
      selectRoom(room);
    });

    if(index%2===0){
      leftRooms.appendChild(el);
    }else{
      rightRooms.appendChild(el);
    }

  });
}


function selectRoom(room){

  if(!running || cleaning) return;

  if(room.status==="ready"){
    showMessage("Номер уже готов ✓");
    return;
  }

  if(room.status==="dnd"){
    changeSatisfaction(7);
    showMessage("⛔ Гость просил не беспокоить!");
    return;
  }

  if(
    room.status==="clean" ||
    room.status==="urgent"
  ){
    cleanRoom(room);
  }
}


function cleanRoom(room){

  cleaning=true;

  moveCart(room);

  const stages=[
    "🛏 Заправляем постель",
    "🛁 Убираем ванную",
    "🗑 Выносим мусор",
    "🧹 Убираем пол",
    "🧴 Проверяем принадлежности"
  ];

  let stage=0;
  let progress=0;

  jobMain.textContent=
    `Номер ${room.number} • ${stages[0]}`;

  jobBar.style.width="0%";

  const timer=setInterval(()=>{

    if(!running){
      clearInterval(timer);
      return;
    }

    progress+=4;

    jobBar.style.width=progress+"%";

    const newStage=
      Math.min(
        Math.floor(progress/20),
        stages.length-1
      );

    if(newStage!==stage){
      stage=newStage;

      jobMain.textContent=
        `Номер ${room.number} • ${stages[stage]}`;
    }

    if(progress>=100){

      clearInterval(timer);

      finishCleaning(room);
    }

  },90);
}


function finishCleaning(room){

  const wasUrgent =
    room.status==="urgent";

  room.status="ready";
  room.waiting=0;

  cleaned++;
  happy++;

  const points =
    wasUrgent ? 200 : 120;

  score+=points;

  /*
  Хорошая работа возвращает
  шкалу к зелёному.
  */
  changeSatisfaction(
    wasUrgent ? -13 : -8
  );

  cleanedEl.textContent=cleaned;
  happyEl.textContent=happy;
  scoreEl.textContent=score;

  jobMain.textContent=
    `Номер ${room.number} готов ✓`;

  jobBar.style.width="100%";

  showMessage(
    wasUrgent
      ? `Отлично! Срочная уборка +${points}`
      : `Гость доволен! +${points}`
  );

  cleaning=false;

  renderRooms();
}


function moveCart(room){

  /*
  4 положения по коридору
  */

  const row =
    Math.floor(room.index/2);

  const positions=[
    7,
    31,
    55,
    79
  ];

  cart.style.top=
    positions[row]+"%";
}


function generateEvent(){

  if(!running) return;

  const readyRooms =
    rooms.filter(
      r=>r.status==="ready"
    );

  if(readyRooms.length===0)
    return;

  const room =
    readyRooms[
      Math.floor(
        Math.random()*readyRooms.length
      )
    ];

  /*
  Иногда сначала появляется
  "Не беспокоить".
  */

  if(Math.random()<0.24){

    room.status="dnd";

    renderRooms();

    setTimeout(()=>{

      if(
        running &&
        room.status==="dnd"
      ){
        room.status="clean";
        room.waiting=0;

        showMessage(
          `🧹 Номер ${room.number} можно убрать`
        );

        renderRooms();
      }

    },3500+Math.random()*3500);

  }else{

    room.status="clean";
    room.waiting=0;

  }

  renderRooms();
}


function updateWaiting(){

  rooms.forEach(room=>{

    if(
      room.status==="clean" ||
      room.status==="urgent"
    ){

      room.waiting+=1.25;

      /*
      После 55% ожидания
      номер становится срочным.
      */

      if(
        room.waiting>=55 &&
        room.status==="clean"
      ){
        room.status="urgent";

        showMessage(
          `⚠️ Номер ${room.number} ждёт слишком долго!`
        );
      }

      /*
      Чем дольше ждёт номер,
      тем быстрее падает довольство.
      */

      if(room.waiting>30)
        changeSatisfaction(.18);

      if(room.waiting>60)
        changeSatisfaction(.25);

      if(room.waiting>=100){

        room.waiting=70;

        changeSatisfaction(6);
      }
    }

  });

  renderRooms();
}


function changeSatisfaction(amount){

  dissatisfaction+=amount;

  dissatisfaction=
    Math.max(
      0,
      Math.min(
        100,
        dissatisfaction
      )
    );

  pointer.style.left=
    dissatisfaction+"%";

  if(dissatisfaction>=72){
    satisfactionBox.classList.add("danger");
  }else{
    satisfactionBox.classList.remove("danger");
  }

  if(dissatisfaction>=100){
    endGame();
  }
}


function updateClock(){

  /*
  +5 игровых минут
  */

  gameMinutes+=5;

  const hours =
    Math.floor(gameMinutes/60)%24;

  const minutes =
    gameMinutes%60;

  clockEl.textContent=
    String(hours).padStart(2,"0")+
    ":"+
    String(minutes).padStart(2,"0");
}


function showMessage(text){

  message.textContent=text;

  clearTimeout(
    showMessage.timer
  );

  showMessage.timer=
    setTimeout(()=>{
      message.textContent="";
    },2300);
}


function startGame(){

  clearInterval(mainTimer);
  clearInterval(eventTimer);

  score=0;
  cleaned=0;
  happy=0;

  dissatisfaction=12;

  gameMinutes=8*60;

  running=true;
  cleaning=false;

  scoreEl.textContent="0";
  cleanedEl.textContent="0";
  happyEl.textContent="0";

  clockEl.textContent="08:00";

  pointer.style.left=
    dissatisfaction+"%";

  satisfactionBox
    .classList.remove("danger");

  jobMain.textContent=
    "Ожидание запросов...";

  jobBar.style.width="0%";

  cart.style.top="48%";

  createRooms();

  startScreen.classList.add("hidden");
  gameOverScreen.classList.add("hidden");

  /*
  Первый запрос появляется
  почти сразу.
  */

  setTimeout(()=>{
    if(running)
      generateEvent();
  },1200);

  /*
  Основной игровой цикл
  */

  mainTimer=setInterval(()=>{

    if(!running) return;

    updateClock();
    updateWaiting();

  },1000);

  /*
  Новые запросы.
  */

  eventTimer=setInterval(()=>{

    if(running)
      generateEvent();

  },4000);
}


function endGame(){

  if(!running) return;

  running=false;
  cleaning=false;

  clearInterval(mainTimer);
  clearInterval(eventTimer);

  document.getElementById(
    "finalCleaned"
  ).textContent=cleaned;

  document.getElementById(
    "finalScore"
  ).textContent=score;

  document.getElementById(
    "finalHappy"
  ).textContent=happy;

  let record=0;

  try{
    record=
      Number(
        localStorage.getItem(
          "madenHotelRecord"
        )
      )||0;

    if(score>record){
      record=score;

      localStorage.setItem(
        "madenHotelRecord",
        record
      );
    }
  }catch(e){}

  document.getElementById(
    "record"
  ).textContent=record;

  gameOverScreen
    .classList.remove("hidden");
}


document.getElementById(
  "startButton"
).addEventListener(
  "click",
  startGame
);

document.getElementById(
  "restartButton"
).addEventListener(
  "click",
  startGame
);


/*
Начальное положение шкалы
*/

pointer.style.left=
  dissatisfaction+"%";

createRooms();

</script>
